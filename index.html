</button>

  color: #0ff; opacity: 1;
  transform: translate(-3px, -2px);

3D CUBE FLIP

HTML
<div class="cube-wrap">
  <button class="cube">
    <span class="front">Hover Me</span>
    <span class="back">Clicked!</span>
  </button>
</div>
CSS
.cube-wrap { perspective: 1000px; }
.cube { transform-style: preserve-3d; transition: .5s; }
.cube:hover { transform: rotateX(-90deg); }
.cube span { position: absolute; inset: 0; }
.cube .front { transform: translateZ(20px); }
.cube .back { transform: rotateX(90deg) translateZ(20px); }
